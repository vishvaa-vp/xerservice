    imageUrl?: string;
}

function formatTime(timeStr?: string | null): string {
    if (!timeStr) return '';
    const [hoursStr, minutesStr] = timeStr.split(':');
    const hours = parseInt(hoursStr, 10);
    const minutes = parseInt(minutesStr, 10);
    if (isNaN(hours)) return timeStr;

    const period = hours >= 12 ? 'PM' : 'AM';
    const displayHours = hours % 12 === 0 ? 12 : hours % 12;
    const displayMinutes = isNaN(minutes) ? '00' : minutes.toString().padStart(2, '0');

    return `${displayHours}:${displayMinutes} ${period}`;
}

function getInitials(name: string): string {
    const parts = name.trim().split(/\s+/).filter(Boolean);
    if (parts.length === 0) return 'XS';
    if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
    return (parts[0][0] + parts[1][0]).toUpperCase();
}

function ShopCard({ shop, onSelect, accentHex }: { shop: HomepageShop; onSelect: (shop: HomepageShop) => void; accentHex: string }) {
    const { user } = useApp();
    const isClosed = shop.status === 'CLOSED';
    const isPaused = shop.status === 'PAUSED';
    const isClosingSoon = shop.status === 'OPEN' && shop.closingSoon;
    const isUnavailable = isClosed || isPaused || isClosingSoon;

    const content = (
        <div
            className="card card-hover"
            style={{
                display: 'grid',
                gridTemplateColumns: '72px minmax(0,1fr)',
                gap: '12px',
                padding: '14px',
                alignItems: 'center',
                opacity: isUnavailable ? 0.68 : 1,
                filter: isUnavailable ? 'grayscale(0.28)' : 'none',
                border: `1px solid ${isClosingSoon || isPaused ? 'var(--accent-border)' : 'var(--border)'}`,
            }}
        >
            <div
                style={{
                    width: '72px',
                    height: '90px',
                    borderRadius: '10px',
                    overflow: 'hidden',
                    background: 'var(--bg-secondary)',
                    border: '1px solid var(--border)',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    flexShrink: 0,
                }}
            >
                {shop.imageUrl ? (
                    <img src={shop.imageUrl} alt={shop.name} style={{ width: '100%', height: '100%', objectFit: 'cover' }} />
                ) : (
                    <span style={{ fontSize: '20px', fontWeight: '900', color: accentHex }}>{shop.imageInitials}</span>
                )}
            </div>

            <div style={{ minWidth: 0 }}>
                <div style={{ display: 'flex', alignItems: 'center', gap: '6px', flexWrap: 'wrap', marginBottom: '4px' }}>
                    <h2 style={{ fontSize: '16px', fontWeight: '900', letterSpacing: 0 }}>{shop.name}</h2>
                    {isClosed ? (
                        <span className="badge badge-outline" style={{ fontSize: '10px', fontWeight: '800', textTransform: 'uppercase' }}>Closed</span>
                    ) : isPaused ? (
                        <span className="badge badge-accent" style={{ fontSize: '10px', fontWeight: '800', textTransform: 'uppercase', background: `${accentHex}22`, color: accentHex, border: `1px solid ${accentHex}55` }}><AlertTriangle size={10} /> Paused</span>
                    ) : isClosingSoon ? (
                        <span className="badge badge-accent" style={{ fontSize: '10px', fontWeight: '800', textTransform: 'uppercase', background: `${accentHex}22`, color: accentHex, border: `1px solid ${accentHex}55` }}><AlertTriangle size={10} /> Closing soon</span>
                    ) : (
                        <span className="badge badge-success" style={{ fontSize: '10px', fontWeight: '800', textTransform: 'uppercase' }}>Open</span>
                    )}
                </div>
                {(isClosingSoon || isPaused) && (
                    <p style={{ fontSize: '11px', color: accentHex, fontWeight: '800', marginBottom: '6px' }}>
                        {shop.closingMessage || (isPaused ? 'Orders are temporarily paused.' : 'Orders paused — closing soon.')}
                    </p>
                )}
                <div style={{ display: 'flex', gap: '10px', flexWrap: 'wrap', marginBottom: '8px' }}>
                    <span style={{ fontSize: '12px', color: 'var(--fg-muted)', fontWeight: '600', display: 'flex', alignItems: 'center', gap: '4px' }}>
                        <Clock size={13} /> {shop.openTime} – {shop.closeTime}
                    </span>
                </div>
                <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', gap: '8px', flexWrap: 'wrap' }}>
                    <div style={{ fontSize: '11px', color: 'var(--fg-muted)', fontWeight: '700' }}>
                        B&W <strong style={{ color: accentHex }}>Rs {shop.priceBwPerPage != null ? shop.priceBwPerPage.toFixed(2) : '—'}</strong> · Color <strong style={{ color: accentHex }}>Rs {shop.priceColorPerPage != null ? shop.priceColorPerPage.toFixed(2) : '—'}</strong>
                    </div>
                    {!isUnavailable ? (
                        <span className="btn btn-sm" style={{ borderRadius: '8px', pointerEvents: 'none', fontSize: '11px', padding: '5px 10px', background: accentHex, color: '#ffffff', border: 'none' }}>
                            <UploadCloud size={13} /><span className="desktop-only">Upload</span><span className="mobile-only">Print Here</span>
                        </span>
                    ) : (
                        <span className="btn btn-outline btn-sm" style={{ borderRadius: '8px', pointerEvents: 'none', opacity: 0.8, fontSize: '11px', padding: '5px 10px' }}>
                            {isClosed ? 'Closed' : 'Paused'}
                        </span>
                    )}
                </div>
            </div>
        </div>
    );

    if (isUnavailable) {
        return <div aria-disabled="true">{content}</div>;
    }

    const shopUploadHref = user ? `/order/upload?shop=${shop.id}` : `/login?redirect=${encodeURIComponent(`/order/upload?shop=${shop.id}`)}`;

    return (
        <Link href={shopUploadHref} onClick={() => onSelect(shop)} style={{ textDecoration: 'none' }}>
            {content}
        </Link>
    );
}
